\newcommand{\letterSender}[6]{%

\newcommand{\letterTo}[2]{#1\if\relax\detokenize{#2}\relax\else\\#2\fi\par\vspace{18pt}}

\newcommand{\letterDateLine}[1]{\textbf{#1}\par}
% \letterSubject{Application for the position of ...}
\newcommand{\letterSubject}[1]{\textbf{#1}\par\vspace{6pt}}
% \letterClosing{Best regards,}{Name}
\newcommand{\letterClosing}[2]{#1\par\vspace{22pt}\rule{5cm}{0.4pt}\\#2}
